cmake_minimum_required(VERSION 3.22)
project(SampleFlip VERSION 0.1.0)
set(CMAKE_CXX_STANDARD 17)
include(FetchContent)
FetchContent_Declare(JUCE GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG 8.0.10 GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(JUCE)
juce_add_plugin(SampleFlip COMPANY_NAME "Sample Flip" BUNDLE_ID com.sampleflip.instrument PLUGIN_MANUFACTURER_CODE SmFl PLUGIN_CODE Sf01 FORMATS VST3 Standalone PRODUCT_NAME "Sample Flip" IS_SYNTH TRUE NEEDS_MIDI_INPUT TRUE IS_MIDI_EFFECT FALSE COPY_PLUGIN_AFTER_BUILD FALSE)
target_sources(SampleFlip PRIVATE Source/Plugin.cpp)
target_compile_definitions(SampleFlip PRIVATE JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0)
target_link_libraries(SampleFlip PRIVATE juce::juce_audio_utils juce::juce_dsp juce::juce_audio_formats juce::juce_recommended_config_flags juce::juce_recommended_warning_flags)
